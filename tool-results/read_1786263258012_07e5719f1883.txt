   343→    fun playerAttack() {
   344→        if (battlePhase != 0) return
   345→        val baseDmg = playerAtk
   346→        val variance = (Math.random() * 6 - 3).toInt()
   347→        val crit = Math.random() < 0.15
   348→        battlePlayerDmg = maxOf(1, baseDmg - battleEnemyAtk / 4 + variance + (if (crit) baseDmg / 2 else 0))
   349→        battleEnemyHp = maxOf(0, battleEnemyHp - battlePlayerDmg)
   350→        battleMessage = if (crit) "${TextData.CRITICAL_HIT} -$battlePlayerDmg" else "Атака! -$battlePlayerDmg"
   351→        battlePhase = 1
   352→        battleAnimTimer = 30
   353→        if (battleEnemyHp <= 0) {
   354→            battlePhase = 3
   355→            battleMessage = TextData.VICTORY
   356→            val expGain = 10 + battleEnemyId * 8
   357→            val goldGain = 5 + battleEnemyId * 3
   358→            playerExp += expGain
   359→            playerGold += goldGain
   360→            battleMessage = "$battleMessage +${expGain} ${TextData.EXP.toLowerCase()} +${goldGain} ${TextData.GOLD.toLowerCase()}"
   361→            checkLevelUp()
   362→            battleActive = false
   363→        }
   364→    }

   366→    fun playerMagic() {
   367→        if (battlePhase != 0 || playerMp < 5) return

   370→        battleEnemyHp = maxOf(0, battleEnemyHp - battlePlayerDmg)
   371→        battleMessage = "Огненный шар! -$battlePlayerDmg"

   374→        if (battleEnemyHp <= 0) {
   375→            battlePhase = 3
   376→            battleMessage = TextData.VICTORY
   377→            val expGain = 10 + battleEnemyId * 8
   378→            playerExp += expGain
   379→            checkLevelUp()
   380→            battleActive = false
   381→        }

   421→        val defending = battleMessage.contains("Защита")
   422→        battleEnemyDmg = maxOf(1, baseDmg - playerDef / 3 + variance - (if (defending) playerDef / 2 else 0))
   423→        playerHp = maxOf(0, playerHp - battleEnemyDmg)
   424→        battleMessage = "${TextData.getEnemyName(battleEnemyId)} атакует! -$battleEnemyDmg ${TextData.HP}"

   426→            battlePhase = 4

   435→    fun checkLevelUp() {
   436→        while (playerExp >= playerExpNext) {
   437→            playerExp -= playerExpNext
   438→            playerLevel++
   439→            playerMaxHp += 10 + playerLevel * 2
   440→            playerMaxMp += 3 + playerLevel
   441→            playerAtk += 2 + playerLevel / 2
   442→            playerDef += 1 + playerLevel / 3
   443→            playerSpd += 1
   444→            playerHp = playerMaxHp
   445→            playerMp = playerMaxMp
   446→            playerExpNext = playerLevel * 30 + playerLevel * playerLevel * 5
   447→            battleMessage += " ${TextData.LEVEL_UP} (${TextData.LVL} $playerLevel)"
   448→        }
   449→    }

   452→    fun movePlayer(dx: Float, dy: Float) {
   453→        if (state != STATE_GAME) return
   454→        var nx = playerX + dx
   455→        var ny = playerY + dy
   456→
   457→        // Direction
   458→        playerDir = when {
   459→            dy > 0 -> 0
   460→            dx < 0 -> 1
   461→            dx > 0 -> 2
   462→            dy < 0 -> 3
   463→            else -> playerDir
   464→        }
   465→
   466→        // Bounds
   467→        nx = nx.coerceIn(0.5f, (mapWidth - 1.5).toFloat())
   468→        ny = ny.coerceIn(0.5f, (mapHeight - 1.5).toFloat())
   469→
   470→        // Walkable check
   471→        val tx = nx.toInt()
   472→        val ty = ny.toInt()
   473→        val idx = ty * mapWidth + tx
   474→        val walk = mapWalkable
   475→        if (walk != null && idx in walk.indices && walk[idx] == 0.toByte()) {
   476→            // Not walkable, check NPC collision
   477→        } else if (walk != null && idx in walk.indices) {
   478→            playerX = nx
   479→            playerY = ny
   480→        }
   481→
   482→        playerMoving = true
   483→
   484→        // Check NPC interaction
   485→        for (npc in npcs) {
   486→ val dist = Math.sqrt(((playerX - npc.x) * (playerX - npc.x) + (playerY - npc.y) * (playerY - npc.y)).toDouble())
   487→            if (dist < 1.5 && (dx != 0f || dy != 0f)) {
   488→                showDialogue(TextData.getNpcDialogue(npc.id))
   489→                // Step back
   490→                playerX -= dx
   491→                playerY -= dy
   492→                return
   493→            }
   494→        }
   495→
   496→        // Random battle encounter
   497→        if (Math.random() < 0.03) {
   498→            val enemies = mapEnemies[currentMapId] ?: intArrayOf(0)
   499→            val eid = enemies[(Math.random() * enemies.size).toInt()]
   500→            startBattle(eid)
   501→        }
   502→
   503→        // Map transition zones
   504→        checkMapTransition()
   505→    }

   507→    private fun checkMapTransition() {
   508→        when (currentMapId) {
   509→            0 -> {
   510→                if (playerX > mapWidth - 2) { loadMap(1); playerX = 1f; }
   511→                if (playerY < 1) { loadMap(4); playerY = (mapHeight - 2).toFloat(); }
   512→            }
   513→            1 -> {
   514→                if (playerX < 1) { loadMap(0); playerX = (mapWidth - 2).toFloat(); }
   515→                if (playerX > mapWidth - 2) { loadMap(2); playerX = 1f; }
   516→            }
   517→            2 -> {
   518→                if (playerX < 1) { loadMap(1); playerX = (mapWidth - 2).toFloat(); }
   519→                if (playerY > mapHeight - 2) { loadMap(3); playerY = 1f; }
   520→            }
   521→            3 -> {
   522→                if (playerY < 1) { loadMap(2); playerY = (mapHeight - 2).toFloat(); }
   523→            }
   524→            4 -> {
   525→                if (playerY > mapHeight - 2) { loadMap(0); playerY = 1f; }
   526→                if (playerX > mapWidth - 2) { loadMap(5); playerX = 1f; }
   527→            }
   528→            5 -> {
   529→                if (playerX < 1) { loadMap(4); playerX = (mapWidth - 2).toFloat(); }
   530→            }
   531→        }
   532→    }

   543→                currentLine = StringBuilder()

   545→            if (currentLine.isNotEmpty()) currentLine.append(" ")
   546→            currentLine.append(word)

   549→        dialogLines = lines

   552→        dialogTimer = 0

   592→    // ---- Update ----
   593→    fun update() {
   594→        frameCounter++
   595→        if (playerMoving) {
   596→            playerFrame = (playerFrame + 1) % 4
   597→        } else {
   598→            playerFrame = 0
   599→        }
   600→        playerMoving = false
   601→
   602→        when (state) {
   603→            STATE_SPLASH -> {
   604→                splashAlpha = minOf(255, splashAlpha + 5)
   605→                if (splashAlpha >= 255) {
   606→                    Thread.sleep(500)
   607→                    goToTitle()
   608→                }
   609→            }
   610→            STATE_BATTLE -> {
   611→                if (battleAnimTimer > 0) {
   612→                    battleAnimTimer--
   613→                    battleShakeX = if (battleAnimTimer > 15) ((Math.random() * 6 - 3).toInt()) else 0
   614→                    battleShakeY = if (battleAnimTimer > 15) ((Math.random() * 4 - 2).toInt()) else 0
   615→                }
   616→                if (battlePhase == 1 && battleAnimTimer <= 0) {
   617→                    battlePhase = 2
   618→                    battleAnimTimer = 20
   619→                }
   620→                if (battlePhase == 2 && battleAnimTimer <= 0) {
   621→                    if (battleActive) enemyTurn()
   622→                }
   623→                if (battleMessageTimer > 0) battleMessageTimer--
   624→            }
   625→            STATE_DIALOG -> {
   626→                dialogTimer++
   627→                if (dialogTimer % 2 == 0 && dialogLineIdx < dialogLines.size) {
   628→                    dialogCharIdx++
   629→                }
   630→            }
   631→        }
   632→
   633→        // Camera follow
   634→        val targetCamX = playerX * TILE_SIZE - screenW / 2f
   635→        val targetCamY = playerY * TILE_SIZE - screenH / 2f
   636→        cameraX += (targetCamX - cameraX) * 0.2f
   637→        cameraY += (targetCamY - cameraY) * 0.2f
   638→        cameraX = cameraX.coerceIn(0f, (mapWidth * TILE_SIZE - screenW).toFloat())
   639→        cameraY = cameraY.coerceIn(0f, (mapHeight * TILE_SIZE - screenH).toFloat())
   640→    }

   716→    private fun renderGame(canvas: Canvas) {
   717→        // Draw map tiles
   718→        val ts = tileset
   719→        val tiles = mapTiles
   720→        if (ts != null && tiles != null) {
   721→            val tsW = ts.width / TILE_SIZE
   722→            val startCol = maxOf(0, (cameraX / TILE_SIZE).toInt())
   723→            val endCol = minOf(mapWidth, ((cameraX + screenW) / TILE_SIZE).toInt() + 1)
   724→            val startRow = maxOf(0, (cameraY / TILE_SIZE).toInt())
   725→            val endRow = minOf(mapHeight, ((cameraY + screenH) / TILE_SIZE).toInt() + 1)
   726→
   727→            val srcRect = Rect()
   728→            val dstRect = RectF()
   729→            for (row in startRow until endRow) {
   730→                for (col in startCol until endCol) {
   731→                    val idx = row * mapWidth + col
   732→                    if (idx >= tiles.size) continue
   733→                    val tileId = tiles[idx].toInt() and 0xFF
   734→                    if (tileId == 0) continue
   735→
   736→                    val srcX = (tileId % tsW) * TILE_SIZE
   737→                    val srcY = (tileId / tsW) * TILE_SIZE
   738→                    val dstX = col * TILE_SIZE - cameraX
   739→                    val dstY = row * TILE_SIZE - cameraY
   740→
   741→                    srcRect.set(srcX, srcY, srcX + TILE_SIZE, srcY + TILE_SIZE)
   742→                    dstRect.set(dstX, dstY, dstX + TILE_SIZE, dstY + TILE_SIZE)
   743→                    canvas.drawBitmap(ts, srcRect, dstRect, null)
   744→                }
   745→            }
   746→        } else {
   747→            // Fallback: draw colored grid
   748→            val startCol = maxOf(0, (cameraX / TILE_SIZE).toInt())
   749→            val endCol = minOf(mapWidth, ((cameraX + screenW) / TILE_SIZE).toInt() + 1)
   750→            val startRow = maxOf(0, (cameraY / TILE_SIZE).toInt())
   751→            val endRow = minOf(mapHeight, ((cameraY + screenH) / TILE_SIZE).toInt() + 1)
   752→            for (row in startRow until endRow) {
   753→                for (col in startCol until endCol) {
   754→                    val idx = row * mapWidth + col
   755→                    val walk = mapWalkable
   756→                    val isWalkable = walk != null && idx < walk.size && walk[idx] != 0.toByte()
   757→                    val color = if (isWalkable) Color.parseColor("#2d5a27") else Color.parseColor("#4a3520")
   758→                    paintBox.color = color
   759→                    val dx = col * TILE_SIZE - cameraX
   760→                    val dy = row * TILE_SIZE - cameraY
   761→                    canvas.drawRect(dx, dy, dx + TILE_SIZE, dy + TILE_SIZE, paintBox)
   762→                }
   763→            }
   764→        }
   765→
   766→        // Draw NPCs
   767→        for (npc in npcs) {
   768→            val nx = npc.x * TILE_SIZE - cameraX
